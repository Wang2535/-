## 二、书籍库完整内容（27+本）

### 数据结构定义

```typescript
interface Book {
  id: string;
  name: string;           // 书名
  author: string;         // 作者（虚构或真实）
  coverTheme: string;     // 封面主题描述（供AI绘图）
  tier: number;           // 所属层级(1-9)
  theme: ThemeCategory;   // 主题分类
  quality: 'common' | 'uncommon' | 'rare'; // 品质
  effectType: EffectType; // 效果类型
  effectDescription: string; // 效果描述
  flavorText: string;     // 风味文本
}

type ThemeCategory =
  | 'virus' | 'network' | 'data' | 'social'
  | 'industrial' | 'mobile' | 'cloud' | 'ai-tech'
  | 'management';

type EffectType =
  | 'buff_combat'      // 战斗增益
  | 'buff_draw'        // 抽牌增益
  | 'buff_resource'    // 资源增益
  | 'passive_info'     // 被动信息
  | 'special_ability'  // 特殊能力
  | 'healing'          // 治疗
  | 'resistance'       // 抗性;
```
